# MPO, not the TrimCI C++ solver -- so there is no compile step in this workflow.

# block2 0.5.3 pins mkl==2021.4 (uv pulls it transitively; stated here for clarity). Its
# wheel bundles the libmkl_avx2/avx512 kernels but NOT libmkl_def, so on the AMD (Zen4)
# qis nodes MKL would reach for the missing generic kernel and die -- the run script sets
# MKL_ENABLE_INSTRUCTIONS=AVX2 to use the bundled AVX2 kernel instead (no lib surgery).
mkl==2021.4
