# Runtime for the block2-DMRG isospectrality-reference HPC shards (Python 3.10).
# build_from_eft needs openfermion (pulls cirq); DMRG needs block2 (Linux manylinux
# wheel, installs clean via uv). The mixed_ci C++ extension is NOT needed here --
# classical.trimci.backend imports it lazily (try/except), and DMRG uses block2's own
# MPO, not the TrimCI C++ solver -- so there is no compile step in this workflow.
numpy==1.26.4
scipy==1.12.0
openfermion==1.6.1
cirq==1.4.0
block2==0.5.3
# block2's manylinux wheel links Intel MKL but does NOT vendor the full runtime --
# it dlopens libmkl_def.so.1 at run time and dies without it. The 2023.x mkl wheel
# ships the .so.1-versioned libs block2 0.5.3 expects (2024+ moved to .so.2); the run
# script adds venv/lib to LD_LIBRARY_PATH so the loader finds them.
mkl==2023.2.0
